export default function Register() {
  const [name, setName] = useState("");
  const [email, setEmail] = useState("");
  const [password, setPassword] = useState("");
  const [error, setError] = useState("");
  const [loading, setLoading] = useState(false);
  const router = useRouter();

  const handleSubmit = async (e: React.FormEvent) => {
    e.preventDefault();
    setLoading(true);
    setError("");

    try {
      const response = await fetch("/api/auth/register", {
        method: "POST",
        headers: {
          "Content-Type": "application/json",
        },
        body: JSON.stringify({ name, email, password }),

      const data = await response.json();

        setError(data.error || "Registration failed");
        return;

      setError("An error occurred");

      setLoading(false);

    <div className="min-h-screen bg-black text-white flex items-center justify-center">
      <div className="max-w-md w-full space-y-8 p-8">
        <div>

        ease: "linear",

          animationProgress.jump(newText.length);
          onAnimationComplete?.();
        },
      });

// Custom hook for animating a single block of text
function useAnimatedText(fullText: string, onAnimationComplete?: () => void) {
    const animationProgress = useMotionValue(0);
    const [visibleCharCount, setVisibleCharCount] = useState(0);

    useEffect(() => {
        if (fullText.length === visibleCharCount) {
            onAnimationComplete?.();
            return;
        }

        const controls = animate(animationProgress, fullText.length, {
            duration: Math.max(0.1, (fullText.length - visibleCharCount) / ANIMATION_CHARS_PER_SECOND),
            ease: "linear",
            onUpdate: (latest) => setVisibleCharCount(Math.floor(latest)),
            onComplete: () => {
                setVisibleCharCount(fullText.length);
                animationProgress.jump(fullText.length);
                onAnimationComplete?.();
            },
        });
        return () => controls.stop();
    }, [fullText, onAnimationComplete]);

    return fullText.slice(0, visibleCharCount);
}

"use client";

import React from "react";
import ReactMarkdown from "react-markdown";
import { ChatAccordion } from "./ChatAccordion";

interface ParagraphAccordion {
  id: string;
  position: number;
  chatMessages: Array<{ role: "user" | "ai"; message: string }>;
  paragraphContext: string;
  userFeedback?: "understood" | "did_not_understand";
}

interface TextRendererProps {
  text: string;
  paragraphAccordions: ParagraphAccordion[];
  currentAccordionId: string | null;
}

export function TextRenderer({
  text,
  paragraphAccordions,
  currentAccordionId,
}: TextRendererProps) {
  const markdownComponents = {
    p: ({ children }) => <div className="mb-4 leading-relaxed">{children}</div>,
    code: ({ children }) => (
      <code className="bg-gray-700 px-2 py-1 rounded text-yellow-300 font-mono text-sm">
        {children}
      </code>
    ),
    pre: ({ children }) => (
      <pre className="bg-gray-800 p-4 rounded mt-2 mb-4 overflow-x-auto border border-gray-600">
        <code className="text-green-300 font-mono text-sm">{children}</code>
      </pre>
    ),
    ul: ({ children }) => (
      <ul className="list-disc list-inside mb-4 ml-4">{children}</ul>
    ),
    ol: ({ children }) => (
      <ol className="list-decimal list-inside mb-4 ml-4">{children}</ol>
    ),
    li: ({ children }) => <li className="mb-1">{children}</li>,
    strong: ({ children }) => {
      const text = children?.toString() || "";
      if (text.includes("✓")) {
        return (
          <span className="font-bold text-green-400 bg-green-900/30 px-2 py-1 rounded">
            {children}
          </span>
        );
      } else if (text.includes("✗")) {
        return (
          <span className="font-bold text-red-400 bg-red-900/30 px-2 py-1 rounded">
            {children}
          </span>
        );
      }
      return <strong className="font-bold text-blue-300">{children}</strong>;
    },
    em: ({ children }) => <em className="italic text-gray-300">{children}</em>,
    h1: ({ children }) => (
      <h1 className="text-2xl font-bold mb-4 text-blue-300">{children}</h1>
    ),
    h2: ({ children }) => (
      <h2 className="text-xl font-bold mb-3 text-blue-300">{children}</h2>
    ),
    h3: ({ children }) => (
      <h3 className="text-lg font-bold mb-2 text-blue-300">{children}</h3>
    ),
  };

  if (paragraphAccordions.length === 0) {
    return (
      <ReactMarkdown components={markdownComponents}>{text}</ReactMarkdown>
    );
  }

  // Filter out the currently active accordion to prevent duplication
  const completedAccordions = paragraphAccordions.filter(
    (accordion) => accordion.id !== currentAccordionId && accordion.userFeedback
  );

  if (completedAccordions.length === 0) {
    return (
      <ReactMarkdown components={markdownComponents}>{text}</ReactMarkdown>
    );
  }

  // Sort accordions by position
  const sortedAccordions = [...completedAccordions].sort(
    (a, b) => a.position - b.position
  );

  let result = [];
  let lastIndex = 0;

  sortedAccordions.forEach((accordion) => {
    // Add text before this accordion
    if (accordion.position > lastIndex) {
      const textChunk = text.slice(lastIndex, accordion.position);
      result.push(
        <ReactMarkdown
          key={`text-${lastIndex}`}
          components={markdownComponents}
        >
          {textChunk}
        </ReactMarkdown>
      );
    }

    // Add the readonly feedback
    if (accordion.userFeedback === "understood") {
      // Simple feedback for "understood"
      result.push(
        <div key={accordion.id} className="my-4 p-3 bg-gray-800 rounded-lg border-l-4 border-green-500">
          <div className="flex items-center gap-2">
            <div className="w-3 h-3 rounded-full bg-green-500"></div>
            <span className="text-sm font-medium text-green-400">
              Your Response: ✓ I understand
            </span>
          </div>
        </div>
      );
    } else {
      // Accordion for "did not understand" with chat history
      result.push(
        <div key={accordion.id} className="my-4">
          <ChatAccordion
            accordionId={accordion.id}
            chatMessages={accordion.chatMessages}
            currentChatInput=""
            setCurrentChatInput={() => {}}
            onSubmit={() => {}}
            isLoadingResponse={false}
            isReadOnly={true}
            userFeedback="did_not_understand"
          />
        </div>
      );
    }

    lastIndex = accordion.position;
  });

  // Add remaining text
  if (lastIndex < text.length) {
    const remainingText = text.slice(lastIndex);
    result.push(
      <ReactMarkdown key={`text-${lastIndex}`} components={markdownComponents}>
        {remainingText}
      </ReactMarkdown>
    );
  }

  return <>{result}</>;
}

                <span className="text-gray-300">Your answer: </span>
                <span
                  className={`font-semibold ${
                    response.isCorrect ? "text-green-400" : "text-red-400"
                  }`}
                >
                  {response.userAnswer}
                </span>
                {!response.isCorrect && (
                  <div className="text-sm">
                    <span className="text-gray-400">Correct answer: </span>
                    <span className="text-green-400 font-medium">
                      {response.correctAnswer}
                    </span>
                  </div>
                )}
              </div>
            </div>
            <div className="text-xs text-gray-500">Question #{index + 1}</div>
          </div>
        ))}
      </div>
    </div>
  );
}
